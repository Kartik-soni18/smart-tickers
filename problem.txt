def build_multi_stock_datasets(data_dir, resolution="micro", ...):
    all_train, all_val, all_test = [], [], []
    for csv_file in sorted(Path(data_dir).glob("*.csv")):
        train_ds, val_ds, test_ds = build_datasets(csv_file, resolution, ...)
        all_train.append(train_ds)
        all_val.append(val_ds)
        all_test.append(test_ds)
    
    return (
        ConcatDataset(all_train),
        ConcatDataset(all_val),
        ConcatDataset(all_test)
    )
Why this matters: Training on one stock = memorizing its noise. Multiple stocks force the model to learn generalizable patterns.

Copy

Problem: zscore computes mean/std over the window itself. In live trading, you don't know the future mean/std of the next 60 bars. Use a rolling z-score from a lookback before start:
Python

Copy
lookback = self.close[start - 200 : start]  # past 200 mins

window = (self.close[start : start + self.window] - mean) / (std + 1e-8)
window = np.clip(window, -3, 3)

class LiveTrader:
    def __init__(self, model_path, symbol, window=60):
        self.model = load_model(model_path)
        self.buffer = deque(maxlen=window + 200)  # lookback + window
    
    def on_tick(self, price, volume):
        self.buffer.append((price, volume))
        if len(self.buffer) >= self.window + 200:
            features = self._prepare(self.buffer)
            prob = self.model(features, return_logits=False)
            return self._act(prob)  # Buy/Sell/Hold with confidence threshold
    
    def _act(self, prob, confidence=0.6):
        if prob.max() > confidence:
            return prob.argmax()  # 0=Sell, 1=Hold, 2=Buy
        return 1  # Hold
