def train_epoch(model, loader, criterion, optimizer, device, epoch, total_epochs):
    model.train()
    total_loss = 0
    correct = 0
    total = 0
    
    for step, (inputs, targets) in enumerate(loader, 1):
        inputs, targets = inputs.to(device), targets.to(device)
        
        optimizer.zero_grad()
        
        # Mixed precision forward
        with autocast():
            outputs = model(inputs)
            loss = criterion(outputs, targets)
        
        # Scaled backward
        scaler.scale(loss).backward()
        scaler.step(optimizer)
        scaler.update()
        
        # Stats (unscale for accurate logging)
        total_loss += loss.item()
        _, predicted = outputs.max(1)
        total += targets.size(0)
        correct += predicted.eq(targets).sum().item()
        
        if step % 100 == 0:
            print(f"  Epoch {epoch}/{total_epochs} | step {step}/{len(loader)} | "
                  f"loss {total_loss/step:.4f} | acc {correct/total:.4f}")
    
    return total_loss / len(loader), correct / total

from collections import Counter
import numpy as np

# Check if your dataset exposes targets
if hasattr(train_ds, 'targets'):
    dist = Counter(train_ds.targets)
elif hasattr(train_ds, 'labels'):
    dist = Counter(train_ds.labels)
else:
    # Extract from DataLoader
    all_labels = []
    for _, labels in train_loader:
        all_labels.extend(labels.numpy())
    dist = Counter(all_labels)

total = sum(dist.values())
print("\n=== CLASS DISTRIBUTION ===")
for cls, count in sorted(dist.items()):
    pct = 100 * count / total
    name = ["Sell", "Hold", "Buy"][cls] if cls < 3 else str(cls)
    print(f"  {name}: {count:,} ({pct:.1f}%)")

if max(dist.values()) / total > 0.7:
    print("⚠️ WARNING: Severe class imbalance detected!")
    print("   Accuracy will be misleading. Use F1-macro or balanced accuracy.")
